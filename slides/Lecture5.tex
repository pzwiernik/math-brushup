\begin{frame}{Spectral decomposition (symmetric case)}
\begin{alertblock}{Theorem (Spectral Theorem)}
If $A\in\R^{n\times n}$ is symmetric, then there exist an orthogonal $U\in\R^{n\times n}$ and a diagonal $\Lambda=\mathrm{diag}(\lambda_1,\dots,\lambda_n)$ such that
\[
A=U\Lambda U^\top,
\quad \lambda_1\ge\lambda_2\ge\cdots\ge\lambda_n.
\]
The columns of $U$ are orthonormal eigenvectors of $A$.
\end{alertblock}
\end{frame}



\begin{frame}{Application: Principal Component Analysis (PCA)} 
\begin{small}
We observe $n$ data points ${\bf x}_1,\ldots,{\bf x}_n \in \R^d$.  
\medskip

\textcolor{SeaGreen}{Step 1 (Centering):} Arrange the data in a matrix $\bs X\in \R^{n\times d}$ with rows $\bs x_1,\ldots,\bs x_n$. Define the centering matrix $H=I_n-\tfrac{1}{n}{\bf 1}{\bf 1}^\top $ and 
then the \alert{centered data} is $H\bs X$.\\
\quad (its rows are the centered data points $\bs x_i-\bar{\bs x}$)
\bigskip

\textcolor{SeaGreen}{Step 2 (Sample covariance):} 
\[
S=\tfrac{1}{n}(H\bs X)^\top  (H\bs X).
\]
\medskip

\textcolor{SeaGreen}{Step 3 (Spectral decomposition):}  
\[
S = U \Lambda U^\top\quad\mbox{ and define } \bs Y=H\bs XU \quad \mbox{(rotate the centered data)}.
\]
The sample covariance of $\bs Y$ is $\tfrac1n \bs Y^\top \bs Y=U^\top  S U=\Lambda$ (diagonal! and so $\bs y_i$ are uncorrelated). \\[3mm]

The columns of $U$ are \textcolor{SeaGreen}{principal directions} and diagonal entries of $\Lambda$ are the \textcolor{SeaGreen}{explained variances}. 
\medskip

Sorting eigenvalues in decreasing order ranks components by variance captured.  
\end{small}
\end{frame}

\begin{frame}{Quadratic forms}
A \textcolor{SeaGreen}{quadratic form} is
\[
Q(\mathbf{x}) = \mathbf{x}^\top A \mathbf{x}, \qquad A \in \R^{n\times n}.
\]

\medskip
\textcolor{SeaGreen}{Why symmetric?}  
If $a_{ij}\neq a_{ji}$, the asymmetric part cancels:
\[
x^\top A x = x^\top \tfrac{1}{2}(A+A^\top)x.
\]
So every quadratic form has a unique symmetric representative.

\medskip
\textcolor{SeaGreen}{Examples:}
\[
Q(x,y) = x^2+5xy+4y^2
\]
comes from
$\begin{pmatrix}1&2\\3&4\end{pmatrix}$
or equivalently
$\begin{pmatrix}1&2.5\\2.5&4\end{pmatrix}$.

\medskip
\textcolor{SeaGreen}{Geometry:} level sets $\{x:Q(x)=c\}$ are ellipses, hyperboloids, linear spaces.
\end{frame}




\begin{frame}{Signs of quadratic forms}
For symmetric $A$:
\begin{itemize}
\item \textcolor{SeaGreen}{Positive definite (PD):} $x^\top A x > 0$ for all $x\neq 0$.
\item \textcolor{SeaGreen}{Negative definite (ND):} $x^\top A x < 0$ for all $x\neq 0$.
\item \textcolor{SeaGreen}{Semidefinite:} always $\ge 0$ (PSD) or $\le 0$ (NSD).
\item \textcolor{SeaGreen}{Indefinite:} can take both signs.
\end{itemize}

\medskip
\textcolor{SeaGreen}{Why it matters:}
\begin{itemize}
\item In optimization, a PD Hessian $\Rightarrow$ strict local minimum.  
\item In economics, concave utility functions $\Rightarrow$ risk aversion.  
\end{itemize}

\medskip
\textcolor{SeaGreen}{Example:}
\[
x^\top
\begin{pmatrix}1&-1\\-1&1\end{pmatrix}
x = (x-y)^2 \ge 0
\]
so the matrix is PSD.
\end{frame}

\begin{frame}{Eigenvalue and minor tests}
 
\textcolor{SeaGreen}{Spectral test (symmetric $A$):}
\begin{itemize}
\item PD $\Longleftrightarrow$ all eigenvalues $>0$;\quad ND $\Longleftrightarrow$ all $<0$;
\item PSD $\Longleftrightarrow$ all $\ge 0$;\quad NSD $\Longleftrightarrow$ all $\le 0$;
\item Indefinite $\Longleftrightarrow$ eigenvalues of different signs.
\end{itemize}
\textcolor{SeaGreen}{Sylvester’s criterion (leading principal minors $D_k$):}
\begin{itemize}
\item PD $\Longleftrightarrow$ $D_k>0$ for $k=1,\ldots,n$;
\item ND $\Longleftrightarrow$ $(-1)^kD_k>0$ for $k=1,\ldots,n$.
\end{itemize}
\textcolor{SeaGreen}{Example:}
\(
A=\begin{pmatrix}-3&2&0\\ 2&-3&0\\ 0&0&-5\end{pmatrix}
\)
has $D_1=-3$, $D_2=5$, $D_3=-25$ $\Rightarrow$ negative definite.
 
\end{frame}




\begin{frame}{Examples of quadratic forms}
\begin{enumerate}
\item
$
A=\begin{pmatrix}1&4&6\\ 4&2&1\\ 6&1&6\end{pmatrix}
$  
$\Rightarrow$ indefinite (mixed signs of minors).

\item
$
A=\begin{pmatrix}3&-1&0\\ -1&2&-1\\ 0&-1&3\end{pmatrix}
$  
$\Rightarrow$ positive definite.

\item
$Q(x,y,z)=xy+yz$  
$\Rightarrow$ indefinite (matrix has both positive and negative eigenvalues).
\end{enumerate}

\medskip
\textcolor{SeaGreen}{Takeaway:} sign of eigenvalues $=$ curvature type.
\end{frame}

\begin{frame}{What is a multivariable function?}
A \textcolor{SeaGreen}{function of \(n\) variables} is a map
\[
f:\ D \subset \R^n \to \R, 
\qquad \mathbf{x}=(x_1,\dots,x_n)\mapsto f(\mathbf{x}),
\]
typically with \(D\) open. When discussing gradients/tangent planes we assume \(f\in C^1\) on \(D\).

\begin{itemize}
\item For \(n=2\): the graph \(z=f(x,y)\) is a \textcolor{SeaGreen}{surface} in \(\R^3\).
\item \textcolor{SeaGreen}{Level curves} (contours): \(\{(x,y)\in D:\ f(x,y)=c\}\).
\item \textcolor{SeaGreen}{Sub/superlevel sets}: \(\{f\le c\},\{f\ge c\}\) — useful for (quasi-)convexity.
\item For \(n>2\): use \textcolor{SeaGreen}{slices or projections} to visualize.
\end{itemize}
\end{frame}

\begin{frame}{Multivariate Gaussian density}
A random vector \(\mathbf{X}\in\R^d\) is \textcolor{SeaGreen}{multivariate normal} with mean \(\mathbf{m}\) and \(\Sigma\succ0\) if
\[
f(\mathbf{x})=(2\pi)^{-d/2}(\det \Sigma)^{-1/2}
\exp\!\left(-\tfrac12(\mathbf{x}-\mathbf{m})^{\top}\Sigma^{-1}(\mathbf{x}-\mathbf{m})\right).
\]
\begin{itemize}
\item Parameters: \(\mathbf{m}=E(\mathbf{X})\), \(\ \Sigma= \mathrm{Cov}(\mathbf{X})\succ0\).\\[4mm]
\item Depends on \textit{Mahalanobis} distance $$\|\mathbf{x}-\mathbf{m}\|_{\Sigma}=\sqrt{(\mathbf{x}-\mathbf{m})^\top\Sigma^{-1}(\mathbf{x}-\mathbf{m})}.$$
\item Level sets are ellipsoids \((\mathbf{x}-\mathbf{m})^\top\Sigma^{-1}(\mathbf{x}-\mathbf{m})=\text{const}\).
\end{itemize}
\end{frame}

\begin{frame}{Partial derivatives}
\begin{block}{Definition}
For \(f:D\subset\R^2\to\R\), the partial derivatives at \((x,y)\) are
\[
f_x(x,y)=\lim_{h\to 0}\frac{f(x+h,y)-f(x,y)}{h},\qquad
f_y(x,y)=\lim_{h\to 0}\frac{f(x,y+h)-f(x,y)}{h},
\]
when these limits exist.	 (We also use more standar notation $\frac{\partial f}{\partial x}(x,y)$, $\frac{\partial f}{\partial y}(x,y)$)
\end{block}
\begin{alertblock}{}
	Equivalently, fix $y$ and define $g(x)=f(x,y)$. Then $f_x(x,y)=g'(x)$.
\end{alertblock}

\textcolor{SeaGreen}{Example (marginal costs):} if
\[
C(x,y)=200+22x+16y^{3/2},
\]
then \(C_x(x,y)=22\) and \(C_y(x,y)=24\sqrt{y}\).

\end{frame}

\begin{frame}[label=LinAp]{The gradient and linear approximation}

For ${\bf x}\in\R^n$, the \textcolor{SeaGreen}{gradient} is
\[
\nabla f({\bf x})\;=\;\big(f_{x_1}({\bf x}),\dots,f_{x_n}({\bf x})\big)\;\in \;\R^n.
\]

\vskip 6pt
\begin{alertblock}{Best linear approximation}	
If $f$ is a $C^1$-function, then for $\bs h\in \R^n$,
\[
f({\bf x}+\bs h)=f({\bf x})+\nabla f({\bf x})^\top \bs h + o(\|\bs h\|).
\]

So the gradient gives the \alert{best linear approximation} of $f$ near ${\bf x}$.
\end{alertblock}
\textcolor{SeaGreen}{Geometry:} \(\nabla f\) is normal to level sets \(f(\mathbf{x})=c\)
\end{frame}

% ------------------------------------------------
\begin{frame}{Why is the gradient the direction of steepest ascent?}

Take $\bs h=t\bs u$ with $\|\bs u\|=1$, $t>0$ small. Then
\[
f({\bf x}+t\bs u) = f({\bf x}) + t\,\langle \nabla f({\bf x}),\bs u\rangle+o(t).
\]

\vskip 6pt
The instantaneous rate of change in direction $\bs u$ is
\[
D_u f({\bf x})\;:=\;\lim_{t\to 0}\frac{f({\bf x}+t\bs u)-f({\bf x})}{t} \;=\; \langle \nabla f({\bf x}),u\rangle.
\]

\vskip 6pt
By Cauchy-Schwarz,
\[
|D_u f({\bf x})| \;\le\; \|\nabla f({\bf x})\|,
\]
with equality if $u$ points in the same direction as $\nabla f({\bf x})$.

\vskip 6pt
\textcolor{SeaGreen}{Conclusion:}  
$\nabla f({\bf x})$ points in the direction of \alert{steepest increase},  
$-\nabla f({\bf x})$ in the direction of \alert{steepest decrease}.
\end{frame}

\begin{frame}{Jacobian and matrix differentiation rules}
Let \(F:\R^n\to\R^m\). The \textcolor{SeaGreen}{Jacobian} at \(\mathbf{x}\) is
\[
{\rm J}F(\mathbf{x})
=\left[\frac{\partial F_i}{\partial x_j}(\mathbf{x})\right]_{i=1,\dots,m;\,j=1,\dots,n}\in\R^{m\times n}.
\]
If \(m=1\), \(F=f\), then \({\rm J}f(\mathbf{x})=\nabla f(\mathbf{x})^\top\).

\medskip
\begin{block}{First-order approximation}
	$$F(\bs x+\bs h)\;=\;F(\bs x)+{\rm J}F(\bs x)\bs h +o(\|\bs h\|).$$
\end{block}

%\textcolor{SeaGreen}{Chain rule:}
%If \(G:\R^m\to\R^p\), \(H=G\circ F\), then
%\[
%{\rm J}H(\mathbf{x})={\rm J}G(F(\mathbf{x}))\,{\rm J}F(\mathbf{x}).
%\]

\medskip
\textcolor{SeaGreen}{Quick identities:}
\begin{enumerate}
\item If \(F(\mathbf{x})=A\mathbf{x}\) with \(A\in\R^{m\times n}\): \({\rm J}F=A\).
\item If \(f(\mathbf{x})=\mathbf{a}^{\top}\mathbf{x}\): \(\nabla f=\mathbf{a}\).
\item If \(f(\mathbf{x})=\mathbf{x}^{\top}A\mathbf{x}\): \(\nabla f=(A+A^{\top})\mathbf{x}\) (\(=2A\mathbf{x}\) if \(A\) symmetric).
\end{enumerate}
\end{frame}


% ------------------------------------------------
\begin{frame}{Higher partial derivatives}
Higher partials are iterated derivatives, e.g.
\[
f_{xy}(x,y)=\frac{\partial}{\partial y}\!\left(\frac{\partial f}{\partial x}(x,y)\right).
\]
\begin{alertblock}{Young's Theorem:}
If \(f_{xy}\) and \(f_{yx}\) are continuous, then \(f_{xy}=f_{yx}\).	We write $\frac{\partial^2 f}{\partial x\partial y}$.
\end{alertblock}
\bigskip

\textcolor{SeaGreen}{Example:} \(f(x,y)=x^2y+e^{xy}\) gives
\[
f_{xy}=2x+e^{xy}(1+xy)=f_{yx}.
\]
\begin{block}{Hessian $\nabla^2 f(\x)\in \R^{n\times n}$}
This is a $n\times n$ matrix defined by $(\nabla^2 f(\x))_{ij}=\frac{\partial^2 f }{\partial x_i \partial x_j}$. (symmetric matrix!)
\end{block}
\qquad If $f(\bs x)=\bs x^\top A\bs x$ with $A$ symmetric, then $\nabla^2f(\bs x)=2A$.
\end{frame}

\begin{frame}{Unconstrained optimization}


\textcolor{SeaGreen}{Local optima.}
\(\mathbf{x}_0\) is a local max (min) if \(f(\mathbf{x})\le (\ge)\ f(\mathbf{x}_0)\) nearby.
If \(f\) is differentiable and \(\mathbf{x}_0\) is an interior local extremum,
\[
\nabla f(\mathbf{x}_0)=\mathbf{0}\quad\text{(first-order condition).}
\]

\medskip
If \(f\in C^2\), Taylor expansion at a stationary point \(\mathbf{x}_0\) gives
\[
f(\mathbf{x}_0+\mathbf{h})=f(\mathbf{x}_0)+\tfrac12\,\mathbf{h}^{\top}( \nabla^2 f(\bs x_0)) \mathbf{h}+o(\|\mathbf{h}\|^2),
\]
so the sign of the quadratic form \(\mathbf{h}^{\top}( \nabla^2 f(\bs x_0)) \mathbf{h}\) determines local behavior.
\end{frame}

\begin{frame}{Local optimality: second-order tests}

Assume \(f\in C^2\) and let \(\nabla^2 f({\bf x})\) be the Hessian matrix.
\vskip 6pt
\textcolor{SeaGreen}{At a stationary point \({\bf x}_0\):}
\begin{itemize}
\item \(\nabla^2 f({\bs x_0})\) positive definite \(\Rightarrow\) local minimum.
\item \(\nabla^2 f({\bs x_0})\) negative definite \(\Rightarrow\) local maximum.
\item \(\nabla^2 f({\bs x_0})\) indefinite \(\Rightarrow\) saddle.
\end{itemize}
\vskip 6pt
\textcolor{SeaGreen}{\(n=2\) test:} Let \(D_2=f_{xx}f_{yy}-f_{xy}^2\) at \({\bf x}_0\).
\[
\begin{array}{l}
D_2>0,\ f_{xx}>0 \Rightarrow \text{local min},\\
D_2>0,\ f_{xx}<0 \Rightarrow \text{local max},\\
D_2<0 \Rightarrow \text{saddle},\quad D_2=0:\ \text{inconclusive}.
\end{array}
\]

\end{frame}

% ------------------------------------------------
\begin{frame}{Examples}

\begin{enumerate}
\item \(f(x,y)=x^2-y^2-xy\). Then \(\nabla f=(2x-y,\,-2y-x)\). The only stationary point is \((0,0)\). The Hessian
\(
\nabla^2 f(0,0)=\begin{pmatrix}2&-1\\ -1&-2\end{pmatrix}
\)
is \textcolor{SeaGreen}{indefinite} \(\Rightarrow\) \((0,0)\) is a \textcolor{SeaGreen}{saddle}.
\item \(f(x,y)=x^2+y^4\). Stationary point: \((0,0)\). 
\(
\nabla^2 f(0,0)=\begin{pmatrix}2&0\\ 0&0\end{pmatrix}
\)
is positive semidefinite; \(f\ge 0\) so \((0,0)\) is a \textcolor{SeaGreen}{global minimum}.
\item \(f(x,y)=x^3+y^3\). Stationary point: \((0,0)\). The Hessian at \((0,0)\) is \(0\); the point is a \textcolor{SeaGreen}{saddle}.
\end{enumerate}

\end{frame}

\begin{frame}{Convex domains}

A set \(D\subset\R^n\) is \textcolor{SeaGreen}{convex} if for any \({\bf x},{\bf y}\in D\) and \(t\in[0,1]\), the point \((1-t){\bf x}+t{\bf y}\in D\).

\end{figure}
\end{frame}

% ------------------------------------------------
\begin{frame}{Convexity, concavity, and global optimality}
\begin{alertblock}{Definition}
On a convex domain \(D\subset\R^n\), \(f\) is \textcolor{SeaGreen}{convex} if
\[
f(\theta\mathbf{x}+(1-\theta)\mathbf{y})\;\le\; \theta f(\mathbf{x})+(1-\theta)f(\mathbf{y}),\quad
\forall\,\mathbf{x},\mathbf{y}\in D,\ \theta\in[0,1].
\]
\textcolor{SeaGreen}{Concave}: reverse the inequality.
\end{alertblock}
\bigskip

\begin{block}{Curvature test (for \(C^2\) functions)}
\(\nabla^2 f(\bs x)\succeq 0\ \forall \mathbf{x}\in D \iff f \text{ convex on }D,\quad
\nabla^2 f(\bs x)\preceq 0 \iff f \text{ concave.}\)
Strict definiteness \(\Rightarrow\) strict convexity/concavity.
\end{block}
\bigskip

\alert{Important fact:} If $f$ is strictly convex (concave) then any local minimum (maximum) is a global minimum (maximum). 
\end{frame}


% ------------------------------------------------
\begin{frame}{Example}

Let \(f(x,y)=2x-y-x^2+xy-y^2\). Then
\[
\nabla f=(2-2x+y,\ -1+x-2y),\qquad
\nabla^2 f(\bs x)=\begin{pmatrix}-2&1\\[2pt]1&-2\end{pmatrix}.
\]
\(\nabla^2 f(\bs x)\) is \textcolor{SeaGreen}{negative definite} \(\Rightarrow\) \(f\) is strictly concave. The unique stationary point solves \(\nabla f={\bf 0}\), giving \((x,y)=(0,1)\), which is a \textcolor{SeaGreen}{global maximum}.

\includegraphics[width=2in]{img/plotxy}
\end{figure}
\end{frame}

\begin{frame}{Economic example: profit maximization}

A firm sells products \(X/Y\) at \(45/55\) euros. Revenue \(R(x,y)=45x+55y\). Cost
\[
C(x,y)=300+x^2+1.5\,y^2-25x-35y.
\]
Profit \(f(x,y)=R(x,y)-C(x,y)\). Then
\[
f_x=-2x+70,\qquad f_y=-3y+90 \ \Rightarrow\ (x^\ast,y^\ast)=(35,30).
\]
Since \(\nabla^2 f(\bs x)=\begin{pmatrix}-2&0\\[2pt]0&-3\end{pmatrix}\) is negative definite everywhere, \(f\) is strictly concave and \((35,30)\) is the \textcolor{SeaGreen}{global maximum}. The maximal profit is \(f(35,30)=2275\).

\end{frame}

\begin{frame}{Least squares as orthogonal projection}
Given data $\bs X\in\R^{n\times d}$ and response $\bs y\in\R^n$, the least-squares estimator solves
\[
\hat\beta=\arg\min_{\beta\in \R^d} \|\bs y-\bs X\beta\|^2,
\qquad
\hat {\bs y}=\bs X\hat\beta.
\]
\textcolor{SeaGreen}{Geometric view (recall Lecture 2):} $\hat {\bs y}$ is the orthogonal projection of $\bs y$ onto the column space of $\bs X$, hence
\[
\bs X^\top (\bs y-\bs X\hat\beta)=\bs 0
\quad\Longleftrightarrow\quad
(\bs X^\top \bs X)\hat\beta=\bs X^\top \bs y.
\]

\medskip
\textcolor{SeaGreen}{Polynomial regression:}  
A common use of least squares is fitting nonlinear trends by expanding the design matrix $\bs X$.  
For instance, with one predictor $x$, we can set
\[
\bs X=\begin{pmatrix}
1 & x_1 & x_1^2 & \cdots & x_1^m\\
\vdots & \vdots & \vdots & & \vdots \\
1 & x_n & x_n^2 & \cdots & x_n^m
\end{pmatrix},
\]
so that the fitted model is $y\approx c_0+c_1 x+\cdots+c_m x^m$.
\end{frame}
% ------------------------------------------------
\begin{frame}{Ridge regression: stabilizing high-variance fits}
When $\bs X^\top \bs X$ is ill-conditioned or $d$ is large, add $\ell_2$ regularization:
\[
\hat\beta_\lambda=\arg\min_\beta \big(\|\bs y-\bs X\beta\|^2+\lambda\|\beta\|^2\big)
\quad\Longrightarrow\quad
\hat\beta_\lambda=(\bs X^\top \bs X+\lambda I)^{-1}\bs X^\top \bs y.
\]
%\textcolor{SeaGreen}{Spectral view:} if $X^\top X=U{\rm diag}(s_1^2,\ldots,s_d^2)U^\top$, then
%\[
%\hat\beta_\lambda
%=\sum_{j=1}^d \frac{s_j}{s_j^2+\lambda}\, u_j\, \langle y,\, X u_j\rangle,
%\]
%so ridge \textcolor{SeaGreen}{shrinks} directions with small $s_j$ (low variance) the most $\big(\tfrac{s_j}{s_j^2+\lambda}<\tfrac1{s_j}\big)$, reducing variance and overfitting.
\end{frame}

% ------------------------------------------------
\begin{frame}{Gradient descent }

Goal: minimize \(f(\bs x)\).
\[
\bs x_{t+1}=\bs x_t-\eta_t\,\nabla f(\bs x_t).
\]
\textbf{Pieces you pick:}
\begin{itemize}
\item \textbf{Step size} \(\eta_t\): constant, diminishing, or via backtracking.
\item \textbf{Stop} when \(\|\nabla f(\bs x_t)\|\) small.
%\item \textbf{Guarantee (sketch):} if \(J\) is convex with \(L\)-Lipschitz gradient, then for \(0<\eta<2/L\) constant, GD converges to a global minimizer.
\end{itemize}
{\footnotesize In practice: a good \(\eta_t\) schedule matter a lot.}

\end{frame}

% ------------------------------------------------
\begin{frame}{GD on least squares (closed form vs iterations)}
Least squares problem has a closed form solution. This still requires inverting a potentially large matrix $\bs X^\top \bs X$. GD gives an alternative  way to find a solution. 
\[
f(\beta)=\frac{1}{n}\|\bs X\beta-{\bf y}\|^2,\qquad
\nabla f(\beta)=\frac{2}{n}\,\bs X^{\top}(\bs X\beta-{\bf y}).
\]
\textbf{GD update:}
\[
\beta_{t+1}=\beta_t-\eta\,\frac{2}{n}\,\bs X^{\top}\big(\bs X\beta_t-{\bf y}\big).
\]
\textbf{Ridge:}
\[
f_{\lambda}(\beta)=\frac{1}{n}\|\bs X\beta-{\bf y}\|^2+\lambda\|\beta\|_2^2,\quad
\nabla f_{\lambda}=\frac{2}{n}\bs X^{\top}(\bs X\beta-{\bf y})+2\lambda\beta.
\]
{\footnotesize Closed form exists \((\bs X^{\top}\bs X)^{-1}\bs X^{\top}{\bf y}\), but GD scales better to huge \(n,p\) or streaming data.}
\end{frame}

% ------------------------------------------------
\begin{frame}{When to use second order methods?}

In general, we update $\bs x_t$ as
$$
\bs x_{t+1}\;:=\;\arg\min f(\bs x_t)+\<\nabla f(\bs x_t),\bs x\>+\tfrac12 (\bs x-\bs x_t)^\top \alert{K} (\bs x-\bs x_t).
$$
If $\alert{K}=I_n$, we recover gradient descent. \\[2mm]
If $\alert{K}=\nabla\nabla^\top f(\bs x_t)$, we get the \textcolor{SeaGreen}{Newton method}. \\[2mm]


\begin{itemize}
\item \textbf{Newton:} \(\bs x_{t+1}=\bs x_t-H^{-1}(\bs x_t)\nabla J(\bs x_t)\) (fast near solution, expensive to form/solve).
\item \textbf{Quasi-Newton (e.g., BFGS, L\!-\!BFGS):} approximate \(H^{-1}\) from gradients only; great for medium scale convex problems.
\item \textbf{Takeaway:} for huge data/models use (S)GD; for smaller smooth convex problems, quasi-Newton shines.
\end{itemize}

\end{frame}

\begin{frame}{Constrained optimization: Lagrange and KKT (teaser)}

We want to maximize/minimize $f(\bs x)$ subject to constraints:  
\[
g_i(\bs x)=0 \quad\text{(equalities)}, 
\qquad
h_j(\bs x)\le 0 \quad\text{(inequalities)}.
\]

\medskip
\textcolor{SeaGreen}{Lagrangian:}
\[
\mathcal{L}(\bs x,\bs \lambda,\bs \mu)
= f(\bs x)+\sum_i \lambda_i g_i(\bs x)+\sum_j \mu_j h_j(\bs x).
\]

\textcolor{SeaGreen}{KKT conditions} (when applicable):
\begin{enumerate}
\item \textbf{Stationarity: } $\nabla_x \mathcal{L}(\bs x^*,\bs \lambda^*,\bs \mu^*)=0$.
\item \textbf{Primal feasibility: } $g_i(\bs x^*)=0,\; h_j(\bs x^*)\le 0$.
\item \textbf{Dual feasibility: } $\mu_j^*\ge 0$.
\item \textbf{Complementary slackness: } $\mu_j^*\,h_j(\bs x^*)=0$.
\end{enumerate}

%\medskip
%\alert{Example (budgeted utility max):} maximize $U(x)$ s.t.\ $p^{\top}x\le B$.  
%At optimum: $\nabla U(x^*)=\mu^*p$, $p^\top x^*\le B$, $\mu^*\ge 0$, and $\mu^*(B-p^\top x^*)=0$.

\end{frame}

% ------------------------------------------------
\begin{frame}{Constrained optimization in 3D (example)}

\textcolor{SeaGreen}{Problem:} maximize
\[
f(x,y,z)=xyz
\quad\text{subject to}\quad
g(x,y,z)=x+y+z-1=0,\;\; x,y,z\ge 0.
\]

\medskip
\textcolor{SeaGreen}{Lagrangian:}
\[
\mathcal{L}(x,y,z,\lambda,\mu_1,\mu_2,\mu_3)
= xyz+\lambda(1-x-y-z)-\mu_1x-\mu_2y-\mu_3z.
\]

\textcolor{SeaGreen}{Stationarity:}
\[
\begin{aligned}
\partial_x \mathcal L &= yz-\lambda-\mu_1=0,\\
\partial_y \mathcal L &= xz-\lambda-\mu_2=0,\\
\partial_z \mathcal L &= xy-\lambda-\mu_3=0.
\end{aligned}
\]

\textcolor{SeaGreen}{Complementary slackness:}  
$\mu_1x=\mu_2y=\mu_3z=0$.

\end{frame}

% ------------------------------------------------
\begin{frame}{Solution and geometry}

At an interior solution ($x,y,z>0$), complementary slackness $\Rightarrow \mu_1=\mu_2=\mu_3=0$.  

\medskip
\textcolor{SeaGreen}{Simplified system:}
\[
\begin{aligned}
yz&=\lambda,\\
xz&=\lambda,\\
xy&=\lambda,\\
x+y+z&=1.
\end{aligned}
\]

\textcolor{SeaGreen}{Deduction:}  
$yz=xz=xy \;\Rightarrow\; x=y=z$.  
Constraint $\Rightarrow\; x=y=z=\tfrac13$.

\medskip
\textcolor{SeaGreen}{Maximizer:} $(\tfrac13,\tfrac13,\tfrac13)$, value $f^\ast=\tfrac1{27}$.

\medskip
\textcolor{SeaGreen}{Geometric intuition:}
\begin{itemize}
\item Constraint $x+y+z=1$ is a plane.
\item Level sets $xyz=c$ are curved surfaces.
\item At optimum: $\nabla f=(yz,xz,xy)$ is parallel to $\nabla g=(1,1,1)$ — the plane just \emph{touches} the surface.
\end{itemize}
\end{frame}
